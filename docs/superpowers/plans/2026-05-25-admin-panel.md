# Admin Panel — Pizzagram Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a password-protected admin panel at `/admin` with an orders viewer (Supabase) and full menu management (data/menu.json).

**Architecture:** Route group `app/admin/(protected)/layout.tsx` guards all admin pages server-side by verifying a JWT `admin_session` cookie. Menu is migrated from a TypeScript static array to `data/menu.json`; `lib/menu-data.ts` provides server-side CRUD; `lib/menu.ts` imports the JSON at bundle time for client components. Admin JWT is signed using `ADMIN_PASSWORD` from `.env.local`.

**Tech Stack:** Next.js 16 App Router, TypeScript, Tailwind CSS (CSS vars `--ink`/`--cream`/`--crimson`/`--smoke`), jose (JWT — already installed), Supabase JS client (already installed), Node.js `fs` module.

```typescript
/**
 * Server-side only. Reads and writes data/menu.json.
 * Never import this in client components (uses Node.js fs).
 */
import { readFileSync, writeFileSync } from "fs";
import path from "path";
import type { MenuItem } from "./menu";

const MENU_PATH = path.join(process.cwd(), "data", "menu.json");

export function readMenu(): MenuItem[] {
  return JSON.parse(readFileSync(MENU_PATH, "utf8")) as MenuItem[];
}

export function writeMenu(items: MenuItem[]): void {
  writeFileSync(MENU_PATH, JSON.stringify(items, null, 2), "utf8");
}

export function getMenuItem(id: string): MenuItem | undefined {
  return readMenu().find((item) => item.id === id);
}

export function addMenuItem(item: MenuItem): void {
  const items = readMenu();
  if (items.find((i) => i.id === item.id)) {
    throw new Error(`Item with id "${item.id}" already exists`);
  }
  writeMenu([...items, item]);
}

export function updateMenuItem(id: string, patch: Partial<MenuItem>): MenuItem {
  const items = readMenu();
  const idx = items.findIndex((i) => i.id === id);
  if (idx === -1) throw new Error(`Item "${id}" not found`);
  const updated = { ...items[idx], ...patch } as MenuItem;
  items[idx] = updated;
  writeMenu(items);
  return updated;
}

export function deleteMenuItem(id: string): void {
  const items = readMenu();
  const filtered = items.filter((i) => i.id !== id);
  if (filtered.length === items.length) throw new Error(`Item "${id}" not found`);
  writeMenu(filtered);
}
```

```typescript
/**
 * JWT helpers for the admin session cookie.
 * Uses ADMIN_PASSWORD (from .env.local) as the signing secret.
 * Cookie name: admin_session, lifetime: 8 hours.
 */
import { SignJWT, jwtVerify } from "jose";

const COOKIE_NAME = "admin_session";
const COOKIE_MAX_AGE = 60 * 60 * 8; // 8 hours

function getSecret(): Uint8Array {
  const pw = process.env.ADMIN_PASSWORD;
  if (!pw) throw new Error("ADMIN_PASSWORD не задан в .env.local");
  return new TextEncoder().encode(pw + ":admin_jwt_v1");
}

export async function signAdminSession(): Promise<string> {
  return new SignJWT({ role: "admin" })
    .setProtectedHeader({ alg: "HS256" })
    .setIssuedAt()
    .setExpirationTime("8h")
    .sign(getSecret());
}

export async function verifyAdminSession(token: string): Promise<boolean> {
  try {
    await jwtVerify(token, getSecret());
    return true;
  } catch {
    return false;
  }
}

/** Reads admin_session cookie from an incoming API Request. */
export async function getAdminSessionFromRequest(req: Request): Promise<boolean> {
  const cookieHeader = req.headers.get("cookie") ?? "";
  const match = cookieHeader.match(
    new RegExp(`(?:^|;\\s*)${COOKIE_NAME}=([^;]+)`)
  );
  if (!match) return false;
  return verifyAdminSession(decodeURIComponent(match[1]));
}

export function buildAdminCookie(token: string): string {
  return `${COOKIE_NAME}=${encodeURIComponent(token)}; HttpOnly; SameSite=Lax; Path=/; Max-Age=${COOKIE_MAX_AGE}`;
}

export function clearAdminCookie(): string {
  return `${COOKIE_NAME}=; HttpOnly; SameSite=Lax; Path=/; Max-Age=0`;
}
```

```typescript
import { NextResponse } from "next/server";
import { timingSafeEqual } from "crypto";
import { signAdminSession, buildAdminCookie } from "@/lib/admin-session";

function constantTimeEqual(a: string, b: string): boolean {
  if (!a || !b) return false;
  const aBuf = Buffer.from(a);
  const bBuf = Buffer.from(b);
  if (aBuf.length !== bBuf.length) return false;
  return timingSafeEqual(aBuf, bBuf);
}

export async function POST(req: Request) {
  let body: { password?: string };
  try {
    body = await req.json();
  } catch {
    return NextResponse.json({ error: "Bad JSON" }, { status: 400 });
  }

  const expected = process.env.ADMIN_PASSWORD ?? "";
  if (!expected) {
    return NextResponse.json({ error: "ADMIN_PASSWORD не задан" }, { status: 500 });
  }

  if (!constantTimeEqual(expected, body.password ?? "")) {
    return NextResponse.json({ error: "Неверный пароль" }, { status: 401 });
  }

  const token = await signAdminSession();
  return NextResponse.json(
    { ok: true },
    { headers: { "Set-Cookie": buildAdminCookie(token) } }
  );
}
```

```typescript
import { NextResponse } from "next/server";
import { getAdminSessionFromRequest } from "@/lib/admin-session";
import { supabase } from "@/lib/supabase";

export async function GET(req: Request) {
  if (!(await getAdminSessionFromRequest(req))) {
    return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
  }

  const { data, error } = await supabase
    .from("orders")
    .select("*, users(phone, name)")
    .order("created_at", { ascending: false })
    .limit(100);

  if (error) {
    console.error("admin orders fetch error:", error);
    return NextResponse.json({ error: "DB error" }, { status: 500 });
  }

  return NextResponse.json({ orders: data ?? [] });
}
```

```typescript
import type { MenuItem } from "@/lib/menu";

export async function GET(req: Request, { params }: { params: Promise<{ id: string }> }) {
  if (!(await getAdminSessionFromRequest(req))) {
    return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
  }
  const { id } = await params;
  const item = getMenuItem(id);
  if (!item) return NextResponse.json({ error: "Not found" }, { status: 404 });
  return NextResponse.json({ item });
}

export async function PATCH(req: Request, { params }: { params: Promise<{ id: string }> }) {
  if (!(await getAdminSessionFromRequest(req))) {
    return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
  }

  let patch: Partial<MenuItem>;
  try {
    patch = await req.json();
  } catch {
    return NextResponse.json({ error: "Bad JSON" }, { status: 400 });
  }

  const { id } = await params;
  try {
    const updated = updateMenuItem(id, patch);
    return NextResponse.json({ ok: true, item: updated });
  } catch (err) {
    return NextResponse.json(
      { error: err instanceof Error ? err.message : "Ошибка" },
      { status: 404 }
    );
  }
}

export async function DELETE(req: Request, { params }: { params: Promise<{ id: string }> }) {
  if (!(await getAdminSessionFromRequest(req))) {
    return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
  }

  const { id } = await params;
  try {
    deleteMenuItem(id);
    return NextResponse.json({ ok: true });
  } catch (err) {
    return NextResponse.json(
      { error: err instanceof Error ? err.message : "Ошибка" },
      { status: 404 }
    );
  }
}
```
